function cmp(a: u8[], b: u8[]): i32 {
  let n: i32 = a.len();
  if (b.len() < n) { n = b.len(); }
  let i: i32 = __mismatch_bytes(a, 0, b, 0, n);
  if (i < n) { return a[i] as i32 - b[i] as i32; }
  return a.len() - b.len();
}

  if (i < n) { return a[i] as i32 - b[i] as i32; }

}
