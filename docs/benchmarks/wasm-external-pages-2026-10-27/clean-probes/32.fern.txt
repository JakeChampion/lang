import "std/i32";
function main(): i32 {
  print("before growth");
  let data: i32[] = [];
  let i: i32 = 0;
  while (i < 32) {
    data = data.append(i);
    i = i + 1;
  }
  i = 0;
  while (i < data.len()) {
    if (data[i] != i) { return 1; }
    i = i + 1;
  }
  print("after growth " + data[data.len() - 1].to_string());
  return 0;
}
