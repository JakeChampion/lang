import "./ir";
import "./util";

function show(ops: ir.Op[]): string {
  let out: string = util.i32_to_string(ops.len()) + ":";
  for op in ops {
    out = out + util.i32_to_string(op.kind_tag) + "," + util.i32_to_string(op.i32_imm)
      + "," + util.i64_to_string(op.i64_imm) + "," + util.i64_to_string(f64_bits(op.f64_imm))
      + "," + util.i32_to_string(op.width) + "," + if (op.unsigned) { "1" } else { "0" }
      + "," + op.str + "," + util.i32_to_string(op.decl) + "," + util.i64_to_string(op.source) + "|";
  }
  return out;
}

function report(ops: ir.Op[], nparams: i32, seed: i32): void {
  let marked: ir.Op[] = [];
  for i in 0..ops.len() {
    marked = marked.append(ir.Op { ...ops[i], decl: i + seed, source: 9000000000i64 + (i as i64) });
  }
  let alias: ir.Op[] = marked;
  let once: ir.Op[] = ir.prune_zero_slot_guards(marked, nparams);
  print(show(once));
  print(show(ir.prune_zero_slot_guards(once, nparams)));
  print(show(alias));
}

function main(): i32 {
  for slot in [0, 1, 7, 31, 255] {
    for seed in 1..11 {
      for nparams in [0, 1, slot + 1] {
        report([], nparams, seed);
        report([ir.op_const_i32(seed), ir.op_store_local(slot), ir.op_load_local(slot), ir.op_call_direct("__fern_rc_is_unique", 1)], nparams, seed);
        report([ir.op_load_local(slot), ir.op_call_direct("__fern_rc_is_unique", 1), ir.op_load_local(slot), ir.op_call_direct("__fern_rc_is_unique", 1)], nparams, seed);
        report([ir.op_block(0), ir.op_loop(0), ir.op_load_local(slot), ir.op_call_direct("__fern_rc_is_unique", 1), ir.op_if(0), ir.op_loop(0), ir.op_load_local(slot), ir.op_call_direct("__fern_rc_is_unique", 1), ir.op_drop(), ir.op_end(), ir.op_else(), ir.op_load_local(slot), ir.op_call_direct("__fern_rc_is_unique", 1), ir.op_drop(), ir.op_end(), ir.op_end(), ir.op_load_local(slot), ir.op_call_direct("__fern_rc_is_unique", 1), ir.op_end()], nparams, seed);
        report([ir.op_if(0), ir.op_loop(0), ir.op_load_local(slot), ir.op_call_direct("__fern_rc_is_unique", 1), ir.op_drop(), ir.op_end(), ir.op_else(), ir.op_load_local(slot), ir.op_call_direct("__fern_rc_is_unique", 1), ir.op_drop(), ir.op_end()], nparams, seed);
        report([ir.op_const_i32(0), ir.op_store_local(slot), ir.op_load_local(slot), ir.op_call_direct("__fern_rc_is_unique", 1), ir.op_if(0), ir.op_end()], nparams, seed);
        report([ir.op_load_local(slot), ir.op_call_direct("__fern_rc_is_unique", 1), ir.op_const_i32(seed), ir.op_store_local(slot), ir.op_load_local(slot), ir.op_call_direct("__fern_rc_is_unique", 1)], nparams, seed);
        report([ir.op_str_slice_frame(slot), ir.op_drop(), ir.op_load_local(slot), ir.op_call_direct("__fern_rc_is_unique", 1)], nparams, seed);
        report([ir.op_load_local(slot), ir.op_call_direct("other", 1), ir.op_load_local(slot)], nparams, seed);
      }
    }
  }
  return 0;
}
