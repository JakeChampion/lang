import "../asmcore";
import "./ssaretainedprogram" as retained;
import "./ssaallocatorisas" as allocators;
import "./ssaarrayboxmachines" as boxes;

function main(): i32 {
  let mod = retained.module();
  if (asmcore.refuse_parse_errors([mod])) {
    return 1;
  }
  let g = retained.program(mod);
  let text = retained.emit(g, g.cache, g.physical, false, true, true, 0, -1).text;
  allocators.run();
  boxes.run(text);
  print("bounded allocation guard controls passed");
  return 0;
}
