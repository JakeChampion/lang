import "../compiler/asmcore";
import "../compiler/asm_ir";
import "../compiler/lexer";
import "../compiler/parser";
import "../compiler/semlower";
import "../compiler/ssaprepare";
import "../compiler/ssacapacityisa" as isa;
import "../compiler/ssaarrayboxmachine" as helper;
import "../compiler/ssaconstructormachine" as machine;
import "std/string";
function main(): i32 {
  let record = "struct Parent { only: i32 } @noinline function parent(value: i32): Parent { return Parent { only: value }; } function main(): i32 { return 0; }";
  let m = parser.parse_module(lexer.tokenize(record));
  let g = semlower.program(semlower.driven(m, "x86-64-linux"));
  let sw = asmcore.Switches { backtrace: false, sanitize: false, leak_check: false, rc_free_debug: false, var_rows: false, shared: false, size: false, harden: false };
  let output = asm_ir.emit_module_ir_unit_prepared(g.im, true, false, "", [], g.im.funcs, g.stab, 0, -1, g.cache, g.physical, semlower.runtime_module_lowering, 0 as usize, sw, true);
  print("parser-error=" + isa.array_box_program(output.text).error);
  print(if (helper.check(output.text)) { "helper=true" } else { "helper=false" });
  print(if (machine.check(output, "parent")) { "constructor=true" } else { "constructor=false" });
  print(if (output.text.contains("cmpq $268435455, %rdi\n    ja __fern_alloc_size_abort.c")) { "size-guard=true" } else { "size-guard=false" });
  return 0;
}
