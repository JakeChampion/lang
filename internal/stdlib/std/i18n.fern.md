# Texts in more than one language

This module gives the texts of a program in the language of the person who uses it.

## Model

A language is a tag of BCP 47, such as `en`, `de-CH` or `zh-Hant-TW`. A person can have a list of languages, the first one the best. On Android, `android.languages` gives that list.

`lookup` chooses one of the languages that a program has. It takes the first tag of the person that a language of the program matches, as the lookup of RFC 4647 does. A tag matches when it is equal to the language with no regard to case. When no language is equal, the tag loses its last subtag and tries again, so `de-CH` takes `de`. When no tag matches, `lookup` gives the default.

A `Strings` has the texts of one language and the texts of a base language, as two stores of `std/kv`. The key of a text is its name. `text` gives the text of a key in the language. When the language has no such text, it gives the text of the base language, or else the key itself. `format` puts its arguments in place of `{0}`, `{1}` and on, in the text of a key.

A count needs a text in the plural form of its language. English has two forms, one and other. Russian has three for whole numbers: one, few and many.

`plural_form` gives the form of a count in a language, with the names of the plural rules of CLDR: `zero`, `one`, `two`, `few`, `many` and `other`. `plural` gives the text of the key and a dot and the form, such as `files.one`, with the count in place of `{n}`. A text with no such form takes the form `other`.

## Contracts

`plural_form` knows the rules for whole numbers of the languages in `<<plural rules>>`. Each other language has the rules of English.

`format` leaves a placeholder with no argument as it is.

## Executable example

```fern test name=std-i18n-example
import "std/i18n";
import "std/kv";

function main(): i32 {
  let en = kv.empty("en.kv").set("hello", "Hello, {0}").set("files.one", "{n} file").set("files.other", "{n} files");
  let de = kv.empty("de.kv").set("hello", "Hallo, {0}").set("files.one", "{n} Datei");
  let lang = i18n.lookup(["en", "de"], ["fr-CA", "de-AT"], "en");
  let s = i18n.strings(lang, de, en);
  if (lang != "de" || s.format("hello", ["Ada"]) != "Hallo, Ada") { return 1; }
  if (s.plural("files", 1i64) != "1 Datei" || s.plural("files", 3i64) != "3 files") { return 2; }
  return 0;
}
```

### `lookup`

```fern
<<lookup>>=
// The language of `available` for the first tag of `wanted` that matches
// one, or `fallback` when none matches.
pub function lookup(available: string[], wanted: string[], fallback: string): string {
  for tag in wanted {
    let t = tag.trim().to_ascii_lower().replace("_", "-");
    while (t.len() > 0) {
      for a in available {
        if (a.to_ascii_lower() == t) {
          return a;
        }
      }
      t = shorter(t);
    }
  }
  return fallback;
}

// The tag `t` with no last subtag. A subtag of one letter, such as the
// `x` of `x-private`, goes with the subtag after it.
function shorter(t: string): string {
  let parts = t.split("-");
  let n = parts.len() - 1;
  if (n > 0 && parts[n - 1].len() == 1) {
    n -= 1;
  }
  let kept: string[] = [];
  for i in 0..n {
    kept = kept.append(parts[i]);
  }
  return kept.join("-");
}

// The language of the tag `t` alone, in lower case: `pt` for `pt-BR`.
pub function language_of(t: string): string {
  return t.trim().to_ascii_lower().replace("_", "-").split("-")[0];
}
```

### `strings`

```fern
<<strings>>=
pub struct Strings { lang: string, texts: kv.Store, base: kv.Store }

// The texts `texts` of the language `lang`, and the texts `base` for the
// keys that `texts` does not have.
pub function strings(lang: string, texts: kv.Store, base: kv.Store): Strings {
  return Strings { lang: lang, texts: texts, base: base };
}

pub function (s: Strings) text(key: string): string {
  if let Some(t) = s.texts.get(key) {
    return t;
  }
  return s.base.text(key, key);
}

// The text of `key` with `args[i]` in place of each `{i}`.
pub function (s: Strings) format(key: string, args: string[]): string {
  return filled(s.text(key), args);
}

function filled(t: string, args: string[]): string {
  let out = t;
  for i in 0..args.len() {
    out = out.replace("{" + i.to_string() + "}", args[i]);
  }
  return out;
}

// The text of `key` in the plural form of `n`, with `n` in place of `{n}`.
pub function (s: Strings) plural(key: string, n: i64): string {
  let form = key + "." + plural_form(s.lang, n);
  let t = s.text(form);
  if (t == form) {
    t = s.text(key + ".other");
  }
  return t.replace("{n}", n.to_string());
}
```

### `plural rules`

The rules are those of CLDR for whole numbers. Each rule takes the count with no sign. `GROUPS` gives the languages of each rule after the first, which is the rule of English.

```fern
<<plural rules>>=
const GROUPS: string[] = ["ja zh ko th vi id ms my lo km", "fr pt hi bn fa", "ru uk be", "sr hr bs", "pl", "cs sk", "ar", "he", "ro", "lt", "lv"];

// The plural form of the whole number `n` in the language of the tag
// `lang`.
pub function plural_form(lang: string, n: i64): string {
  let rules: ((i64) => string)[] = [english, other_only, zero_or_one, east_slavic, south_slavic, polish, czech, arabic, hebrew, romanian, lithuanian, latvian];
  return rules[rule_of(language_of(lang))](n.abs());
}

function rule_of(l: string): i32 {
  for i in 0..GROUPS.len() {
    if (GROUPS[i].split(" ").contains(l)) {
      return i + 1;
    }
  }
  return 0;
}

function english(v: i64): string {
  return if (v == 1i64) { "one" } else { "other" };
}

function other_only(v: i64): string {
  return "other";
}

function zero_or_one(v: i64): string {
  return if (v <= 1i64) { "one" } else { "other" };
}

// The last digit is 2, 3 or 4, and the last two digits are not 12 to 14.
function slavic_few(v: i64): boolean {
  let ten = v % 10i64;
  let hundred = v % 100i64;
  return ten >= 2i64 && ten <= 4i64 && (hundred < 12i64 || hundred > 14i64);
}

function slavic_one(v: i64): boolean {
  return v % 10i64 == 1i64 && v % 100i64 != 11i64;
}

function east_slavic(v: i64): string {
  if (slavic_one(v)) {
    return "one";
  }
  return if (slavic_few(v)) { "few" } else { "many" };
}

function south_slavic(v: i64): string {
  if (slavic_one(v)) {
    return "one";
  }
  return if (slavic_few(v)) { "few" } else { "other" };
}

function polish(v: i64): string {
  if (v == 1i64) {
    return "one";
  }
  return if (slavic_few(v)) { "few" } else { "many" };
}

function czech(v: i64): string {
  if (v == 1i64) {
    return "one";
  }
  return if (v >= 2i64 && v <= 4i64) { "few" } else { "other" };
}

function arabic(v: i64): string {
  let hundred = v % 100i64;
  if (v <= 2i64) {
    return ["zero", "one", "two"][v as i32];
  }
  if (hundred >= 3i64 && hundred <= 10i64) {
    return "few";
  }
  return if (hundred >= 11i64) { "many" } else { "other" };
}

function hebrew(v: i64): string {
  if (v == 1i64) {
    return "one";
  }
  return if (v == 2i64) { "two" } else { "other" };
}

function romanian(v: i64): string {
  let hundred = v % 100i64;
  if (v == 1i64) {
    return "one";
  }
  return if (v == 0i64 || hundred >= 1i64 && hundred <= 19i64) { "few" } else { "other" };
}

// The last two digits are not 11 to 19.
function not_teen(v: i64): boolean {
  let hundred = v % 100i64;
  return hundred < 11i64 || hundred > 19i64;
}

function lithuanian(v: i64): string {
  let ten = v % 10i64;
  if (ten == 1i64 && not_teen(v)) {
    return "one";
  }
  return if (ten >= 2i64 && not_teen(v)) { "few" } else { "other" };
}

function latvian(v: i64): string {
  let ten = v % 10i64;
  if (ten == 0i64 || !not_teen(v)) {
    return "zero";
  }
  return if (ten == 1i64 && v % 100i64 != 11i64) { "one" } else { "other" };
}
```
