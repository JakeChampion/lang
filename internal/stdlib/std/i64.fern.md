# 64-bit integer methods

This module provides signed 64-bit integer methods.

## Model

The module extends signed 64-bit values with arithmetic, number properties, bit operations, and text conversion.

## Contracts

Checked operations return an optional result. Saturation operations bound the result. Ordinary operations retain the language arithmetic rules.

## Executable example

```fern test name=std-i64-example
import "std/i64";

function main(): i32 {
  let n: i64 = 48;
  if (n.gcd(18) != 6 || n.to_string() != "48") { return 1; }
  return 0;
}
```

```fern
pub function (n: i64) pluralize(singular: string, plural: string): string {
  return n.unsigned_abs().pluralize(singular, plural);
}

// The whole percentage n is of total, truncated toward zero; 0 when total <= 0.
pub function (n: i64) percent_of(total: i64): i64 {
  if (total <= 0 as i64) {
    return 0 as i64;
  }
  let p: i64 = n.unsigned_abs().percent_of(total as u64) as i64;
  if (n < 0 as i64) {
    return 0 as i64 - p;
  }
  return p;
}

// The largest multiple of m at or below n; n itself when m <= 0.
pub function (n: i64) round_down_to(m: i64): i64 {
  if (m <= 0 as i64) {
    return n;
  }
  return n - n.rem_euclid(m);
}
```
