# Binary32 numbers

This module has the `f32` methods, and it parses decimal text directly to binary32. Parse failures retain their reason and byte offset.

## Executable example

```fern test name=f32-example
import "std/f32";

function main(): i32 {
  let Ok(value) = f32.parse_f32("-0") else { return 1; };
  if (f32_bits(value) != -2147483647 - 1) { return 2; }
  return 0;
}
```

### Module implementation

```fern
<<module implementation>>=
// std/f32 — binary32 methods and numeric conversion.

import "std/number_parse";
import "std/f64";
import "core/f64_text";

// Parse directly to binary32, with an exact check at rounding midpoints.
pub function parse_f32(s: str): Result[f32, number_parse.ParseNumberError] {
  let (value, code) = number_parse._float32(s);
  if (code != number_parse.SUCCESS) {
    return Err(number_parse._error(code, 10));
  }
  return Ok(value);
}
```

### `limits`

The type's limits and special values as constants, read `f32.MAX` or `f32.NAN` after `import "std/f32";`. `EPSILON` is the gap between 1.0 and the next `f32`, and `MIN_POSITIVE` the smallest normal value.

```fern
<<limits>>=
pub const NAN: f32 = 0.0 / 0.0;

pub const INFINITY: f32 = 1.0 / 0.0;

pub const NEG_INFINITY: f32 = -1.0 / 0.0;

pub const EPSILON: f32 = 1.1920929e-7;

pub const MAX: f32 = 3.4028235e38;

pub const MIN: f32 = -3.4028235e38;

pub const MIN_POSITIVE: f32 = 1.17549435e-38;
```

pub function (x: f32) hypot(y: f32): f32 {
  let ax = (x as f64).abs();
  let ay = (y as f64).abs();
  let hi = ax.max(ay);
  let lo = ax.min(ay);
  if (hi == 0.0) {
    return 0.0 as f32;
  }
  let r = lo / hi;
  return (hi * (1.0 + r * r).sqrt()) as f32;
}

  if (x != x) {
    return x;

  if (x > 0.0 as f32) {
    return 1.0 as f32;
  }
  if (x < 0.0 as f32) {
    return 0.0 as f32 - 1.0 as f32;
  }

pub function (a: f32) lerp(b: f32, t: f32): f32 {
  return a + (b - a) * t;
}

}

pub function (x: f32) copysign(sign: f32): f32 {
  let m = x.abs();
  if (sign < 0.0 as f32) {
    return 0.0 as f32 - m;

  return m;
}

pub function (a: f32) midpoint(b: f32): f32 {
  return (a as f64 * 0.5 + b as f64 * 0.5) as f32;
