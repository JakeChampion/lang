# Unsigned 32-bit integer methods

This module provides unsigned 32-bit integer methods.

## Model

Unsigned integer methods cover numeric properties, bit operations, arithmetic, and decimal text. All values lie in the 32-bit unsigned range.

## Contracts

Checked, saturation, and ordinary methods express different overflow policies. Signed interpretations require an explicit conversion.

## Executable example

```fern test name=std-u32-example
import "std/u32";

function main(): i32 {
  let n: u32 = 12;
  if (!n.is_even() || n.to_string() != "12") { return 1; }
  return 0;
}
```

```fern
pub function (n: u32) checked_pow(exp: i32): Option[u32] {
  if (exp < 0) {
    return None;
  }
  let result: u64 = 1 as u64;
  let base: u64 = n as u64;
  let e: i32 = exp;
  while (e > 0) {
    if (e % 2 == 1) {
      result = result * base;
      if (result > 4294967295 as u64) {
        return None;
      }
    }
    e = e / 2;
    if (e > 0) {
      base = base * base;
      if (base > 4294967295 as u64) {
        return None;
      }
    }
  }
  return Some(result as u32);
}
```

```fern
pub function (n: u32) count_zeros(): i32 {
  return 32 - n.count_ones();
}

pub function (n: u32) leading_zeros(): i32 {
  return __clz32(n);
}

pub function (n: u32) trailing_zeros(): i32 {
  return __ctz32(n);
}

pub function (n: u32) bit_length(): i32 {
  return 32 - n.leading_zeros();
}

pub function (n: u32) rotate_left(bits: i32): u32 {
  let k: i32 = bits & 31;
  if (k == 0) {
    return n;
  }
  let left: u32 = n << k as u32;
  let right: u32 = n >> (32 - k) as u32;
  return left | right;
}

pub function (n: u32) rotate_right(bits: i32): u32 {
  let k: i32 = bits & 31;
  if (k == 0) {
    return n;
  }
  let right: u32 = n >> k as u32;
  let left: u32 = n << (32 - k) as u32;
  return right | left;
}

pub function (n: u32) byte_swap(): u32 {
  let b0: u32 = n & 255 as u32;
  let b1: u32 = n >> 8 as u32 & 255 as u32;
  let b2: u32 = n >> 16 as u32 & 255 as u32;
  let b3: u32 = n >> 24 as u32 & 255 as u32;
  return b0 << 24 as u32 | b1 << 16 as u32 | b2 << 8 as u32 | b3;
}

pub function (n: u32) reverse_bits(): u32 {
  let u: u32 = n;
  let r: u32 = 0 as u32;
  let i: i32 = 0;
  while (i < 32) {
    r = r << 1 as u32 | u & 1 as u32;
    u = u >> 1 as u32;
    i = i + 1;
  }
  return r;
}

pub function (n: u32) bit(i: i32): boolean {
  if (i < 0 || i >= 32) {
    return false;
  }
  return (n & 1 as u32 << i as u32) != 0 as u32;
}

pub function (n: u32) set_bit(i: i32): u32 {
  if (i < 0 || i >= 32) {
    return n;
  }
  return n | 1 as u32 << i as u32;
}

pub function (n: u32) clear_bit(i: i32): u32 {
  if (i < 0 || i >= 32) {
    return n;
  }
  let inv: u32 = (0 - 1) as u32 ^ 1 as u32 << i as u32;
  return n & inv;
}

pub function (n: u32) toggle_bit(i: i32): u32 {
  if (i < 0 || i >= 32) {
    return n;
  }
  return n ^ 1 as u32 << i as u32;
}

pub function (n: u32) div_ceil(d: u32): u32 {
  if (d == 0) {
    return 0;
  }
  if (n % d == 0) {
    return n / d;
  }
  return n / d + 1;
}

pub function (n: u32) div_floor(d: u32): u32 {
  if (d == 0) {
    return 0;
  }
  return n / d;
}

pub function (n: u32) next_multiple_of(d: u32): u32 {
  if (d == 0) {
    return 0;
  }
  return n.div_ceil(d) * d;
}

pub function (n: u32) is_multiple_of(d: u32): boolean {
  if (d == 0) {
    return false;
  }
  return n % d == 0;
}
```

```fern
pub function parse_u32_radix(s: str, radix: i32): Result[u32, number_parse.ParseNumberError] {
```
