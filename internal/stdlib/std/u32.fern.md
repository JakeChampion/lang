# Unsigned 32-bit integer methods

This module provides unsigned 32-bit integer methods.

## Model

Unsigned integer methods cover numeric properties, bit operations, arithmetic, and decimal text. All values lie in the 32-bit unsigned range.

## Contracts

Checked, saturation, and ordinary methods express different overflow policies. Signed interpretations require an explicit conversion.

## Executable example

```fern test name=std-u32-example
import "std/u32";

function main(): i32 {
  let n: u32 = 12;
  if (!n.is_even() || n.to_string() != "12") { return 1; }
  return 0;
}
```

```fern
pub function (n: u32) pow(exp: i32): u32 {
  if (exp < 0) {
    return 0 as u32;
  }
  let base: u32 = n;
  let e: i32 = exp;
  let out: u32 = 1 as u32;
  while (e > 0) {
    if (e % 2 == 1) {
      out = out * base;
    }
    base = base * base;
    e = e / 2;
  }
  return out;
}
```

### `4 pub function (n: u32) saturating_mul(other: u32): u32`

```fern
pub function (n: u32) saturating_mul(other: u32): u32 {
  let p: u64 = n as u64 * (other as u64);
  if (p > 4294967295 as u64) {
    return 4294967295 as u32;
  }
  return p as u32;
}

pub function (n: u32) saturating_div(other: u32): u32 {
  return n / other;
}

pub function (n: u32) checked_pow(exp: i32): Option[u32] {
  if (exp < 0) {
    return None;
  }
  let result: u64 = 1 as u64;
  let base: u64 = n as u64;
  let e: i32 = exp;
  while (e > 0) {
    if (e % 2 == 1) {
      result = result * base;
      if (result > 4294967295 as u64) {
        return None;
      }
    }
    e = e / 2;
    if (e > 0) {
      base = base * base;
      if (base > 4294967295 as u64) {
        return None;
      }
    }
  }
  return Some(result as u32);
}

pub function (n: u32) overflowing_add(other: u32): (u32, boolean) {
  let r: u32 = n + other;
  return (r, r < n);
}

pub function (n: u32) overflowing_sub(other: u32): (u32, boolean) {
  let r: u32 = n - other;
  return (r, other > n);
}

pub function (n: u32) overflowing_mul(other: u32): (u32, boolean) {
  let p: u64 = n as u64 * (other as u64);
  return (p as u32, p > 4294967295 as u64);
}

pub function (n: u32) overflowing_div(other: u32): (u32, boolean) {
  return (n / other, false);
}

pub function (n: u32) overflowing_rem(other: u32): (u32, boolean) {
  return (n % other, false);
}

pub function (n: u32) checked_shl(rhs: i32): Option[u32] {
  return n <<? rhs as u32;
}

pub function (n: u32) checked_shr(rhs: i32): Option[u32] {
  return n >>? rhs as u32;
}

pub function (n: u32) sqrt_floor(): u32 {
  if (n < 2 as u32) {
    return n;
  }
  let x: u32 = n / (2 as u32);
  while (true) {
    let y: u32 = (x + n / x) / (2 as u32);
    if (y >= x) {
      return x;
    }
    x = y;
  }
  return x;
}

pub function (n: u32) is_power_of_2(): boolean {
  if (n == 0 as u32) {
    return false;
  }
  return (n & n - 1 as u32) == 0 as u32;
}

pub function (n: u32) next_power_of_2(): u32 {
  if (n <= 1 as u32) {
    return 1 as u32;
  }
  if (n > 2147483648 as u32) {
    return 0 as u32;
  }
  let p: u32 = 1 as u32;
  while (p < n) {
    p = p * (2 as u32);
  }
  return p;
}

pub function (n: u32) log2_floor(): i32 {
  if (n == 0 as u32) {
    return 0 - 1;
  }
  let r: i32 = 0;
  let m: u32 = n;
  while (m > 1 as u32) {
    m = m / (2 as u32);
    r = r + 1;
  }
  return r;
}

pub function (n: u32) log10_floor(): i32 {
  if (n == 0 as u32) {
    return 0 - 1;
  }
  let r: i32 = 0;
  let m: u32 = n;
  while (m >= 10 as u32) {
    m = m / (10 as u32);
    r = r + 1;
  }
  return r;
}

pub function (n: u32) count_ones(): i32 {
  return __popcount32(n);
}
```

### `5 pub function (n: u32) count_zeros(): i32`

```fern
<<5 pub function (n: u32) count_zeros(): i32>>=
pub function (n: u32) count_zeros(): i32 {
  return 32 - n.count_ones();
}

pub function (n: u32) leading_zeros(): i32 {
  return __clz32(n);
}

pub function (n: u32) trailing_zeros(): i32 {
  return __ctz32(n);
}

pub function (n: u32) bit_length(): i32 {
  return 32 - n.leading_zeros();
}

pub function (n: u32) rotate_left(bits: i32): u32 {
  let k: i32 = bits & 31;
  if (k == 0) {
    return n;
  }
  let left: u32 = n << k as u32;
  let right: u32 = n >> (32 - k) as u32;
  return left | right;
}

pub function (n: u32) rotate_right(bits: i32): u32 {
  let k: i32 = bits & 31;
  if (k == 0) {
    return n;
  }
  let right: u32 = n >> k as u32;
  let left: u32 = n << (32 - k) as u32;
  return right | left;
}

pub function (n: u32) byte_swap(): u32 {
  let b0: u32 = n & 255 as u32;
  let b1: u32 = n >> 8 as u32 & 255 as u32;
  let b2: u32 = n >> 16 as u32 & 255 as u32;
  let b3: u32 = n >> 24 as u32 & 255 as u32;
  return b0 << 24 as u32 | b1 << 16 as u32 | b2 << 8 as u32 | b3;
}

pub function (n: u32) reverse_bits(): u32 {
  let u: u32 = n;
  let r: u32 = 0 as u32;
  let i: i32 = 0;
  while (i < 32) {
    r = r << 1 as u32 | u & 1 as u32;
    u = u >> 1 as u32;
    i = i + 1;
  }
  return r;
}

pub function (n: u32) bit(i: i32): boolean {
  if (i < 0 || i >= 32) {
    return false;
  }
  return (n & 1 as u32 << i as u32) != 0 as u32;
}

pub function (n: u32) set_bit(i: i32): u32 {
  if (i < 0 || i >= 32) {
    return n;
  }
  return n | 1 as u32 << i as u32;
}

pub function (n: u32) clear_bit(i: i32): u32 {
  if (i < 0 || i >= 32) {
    return n;
  }
  let inv: u32 = (0 - 1) as u32 ^ 1 as u32 << i as u32;
  return n & inv;
}

pub function (n: u32) toggle_bit(i: i32): u32 {
  if (i < 0 || i >= 32) {
    return n;
  }
  return n ^ 1 as u32 << i as u32;
}

pub function (n: u32) div_ceil(d: u32): u32 {
  if (d == 0) {
    return 0;
  }
  if (n % d == 0) {
    return n / d;
  }
  return n / d + 1;
}

pub function (n: u32) div_floor(d: u32): u32 {
  if (d == 0) {
    return 0;
  }
  return n / d;
}

pub function (n: u32) next_multiple_of(d: u32): u32 {
  if (d == 0) {
    return 0;
  }
  return n.div_ceil(d) * d;
}

pub function (n: u32) is_multiple_of(d: u32): boolean {
  if (d == 0) {
    return false;
  }
  return n % d == 0;
}
```

```fern
pub function parse_u32_radix(s: str, radix: i32): Result[u32, number_parse.ParseNumberError] {
  let (bits, code) = number_parse._integer(s, radix, false, 4294967295 as u64, 0 as u64);
  if (code != number_parse.SUCCESS) {
    return Err(number_parse._error(code, radix));

  return Ok(bits as u32);
}

  if (base == 10) {
    return n.to_string();

}
```

### `digits and number theory`

Digit and number-theory methods that `std/i32` already has, without the signed-only ones (`abs`, `signum`, `sign_str`, `is_minus_one`).

```fern
<<digits and number theory>>=
pub function (n: u32) divmod(d: u32): (u32, u32) {
  return (n / d, n % d);
}

pub function (n: u32) is_one(): boolean {
  return n == 1 as u32;
}

pub function (n: u32) gcd(other: u32): u32 {
  let a: u32 = n;
  let b: u32 = other;
  while (b != 0 as u32) {
    let t: u32 = b;
    b = a % b;
    a = t;
  }
  return a;
}

pub function (n: u32) lcm(other: u32): u32 {
  if (n == 0 as u32 || other == 0 as u32) {
    return 0 as u32;
  }
  return n / n.gcd(other) * other;
}

pub function (n: u32) is_perfect_square(): boolean {
  let k: u32 = n.sqrt_floor();
  return k * k == n;
}

pub function (n: u32) is_prime(): boolean {
  if (n < 2 as u32) {
    return false;
  }
  if (n < 4 as u32) {
    return true;
  }
  if (n % (2 as u32) == 0 as u32 || n % (3 as u32) == 0 as u32) {
    return false;
  }
  let i: u32 = 5 as u32;
  while (i <= n / i) {
    if (n % i == 0 as u32 || n % (i + 2 as u32) == 0 as u32) {
      return false;
    }
    i = i + 6 as u32;
  }
  return true;
}

pub function (n: u32) digits(): i32 {
  let m: u32 = n;
  let c: i32 = 1;
  while (m >= 10 as u32) {
    c = c + 1;
    m = m / (10 as u32);
  }
  return c;
}

pub function (n: u32) has_digit(d: i32): boolean {
  if (d < 0 || d > 9) {
    return false;
  }
  let m: u32 = n;
  while (true) {
    if ((m % (10 as u32)) as i32 == d) {
      return true;
    }
    m = m / (10 as u32);
    if (m == 0 as u32) {
      return false;
    }
  }
  return false;
}

pub function (n: u32) sum_of_digits(): i32 {
  let m: u32 = n;
  let s: i32 = 0;
  while (m > 0 as u32) {
    s = s + (m % (10 as u32)) as i32;
    m = m / (10 as u32);
  }
  return s;
}

// Wraps when the reversed number does not fit.
pub function (n: u32) reverse_digits(): u32 {
  let m: u32 = n;
  let r: u32 = 0 as u32;
  while (m > 0 as u32) {
    r = r * (10 as u32) + m % (10 as u32);
    m = m / (10 as u32);
  }
  return r;
}

pub function (n: u32) is_palindrome(): boolean {
  return n == n.reverse_digits();
}

// Zero above 12, where the result no longer fits.
pub function (n: u32) factorial(): u32 {
  if (n > 12 as u32) {
    return 0 as u32;
  }
  let result: u32 = 1 as u32;
  let i: u32 = 2 as u32;
  while (i <= n) {
    result = result * i;
    i = i + 1 as u32;
  }
  return result;
}

pub function (n: u32) ordinal(): string {
  let suffix: string = "th";
  let mod100: i32 = (n % (100 as u32)) as i32;
  if (mod100 < 11 || mod100 > 13) {
    let mod10: i32 = mod100 % 10;
    if (mod10 == 1) {
      suffix = "st";
    } else if (mod10 == 2) {
      suffix = "nd";
    } else if (mod10 == 3) {
      suffix = "rd";
    }
  }
  return n.to_string() + suffix;
}

pub function (n: u32) pluralize(singular: string, plural: string): string {
  if (n == 1 as u32) {
    return singular;
  }
  return plural;
}

// The whole percentage n is of total, truncated; 0 when total is 0.
pub function (n: u32) percent_of(total: u32): u32 {
  if (total == 0) {
    return 0;
  }
  return (n as u64 * (100 as u64) / (total as u64)) as u32;
}

// The largest multiple of m at or below n; n itself when m is 0.
pub function (n: u32) round_down_to(m: u32): u32 {
  if (m == 0 as u32) {
    return n;
  }
  return n - n % m;
}

pub function gcd_all(xs: u32[]): u32 {
  let g: u32 = 0 as u32;
  for x in xs {
    g = g.gcd(x);
  }
  return g;
}

pub function lcm_all(xs: u32[]): u32 {
  let l: u32 = 1 as u32;
  for x in xs {
    l = l.lcm(x);
  }
  return l;
}
```
