# WebAssembly variable-length integers

This module encodes variable-length integers for WebAssembly.

## Model

Unsigned and signed LEB128 encode integers in seven-bit groups. Each continuation bit indicates another byte, and signed termination preserves the sign.

## Contracts

The selected helper fixes signedness and integer width. Size helpers describe the unsigned encoded length.

## Executable example

```fern test name=std-wasm-leb128-example
import "std/wasm/leb128";

function main(): i32 {
  let bytes = leb128.uleb_u32([], 624485);
  if (bytes.len() != 3 || bytes[0] != 229 as u8 || bytes[1] != 142 as u8 || bytes[2] != 38 as u8) { return 1; }
  return 0;
}
```

```fern
<<2 pub function uleb_u32(buf: u8[], value: u32): u8[]>>=
pub function uleb_u32(buf: u8[], value: u32): u8[] {
  let out: u8[] = buf;
  let v: u32 = value;
  while (true) {
    let lo: u32 = v & 127u32;
    v = v >> 7;
    if (v == 0u32) {
      out = out.append(lo as u8);
      return out;
    }
    out = out.append((lo | 128u32) as u8);
  }
  return out;
}

pub function uleb_u64(buf: u8[], value: u64): u8[] {
  let out: u8[] = buf;
  let v: u64 = value;
  while (true) {
    let lo: u64 = v & 127u64;
    v = v >> 7;
    if (v == 0u64) {
      out = out.append(lo as u8);
      return out;
    }
    out = out.append((lo | 128u64) as u8);
  }
  return out;
}

pub function sleb_i32(buf: u8[], value: i32): u8[] {
  let out: u8[] = buf;
  let v: i32 = value;
  while (true) {
    let byte: i32 = v & 127;
    v = v >> 7;
    let sign_bit_set: boolean = (byte & 64) != 0;
    let done: boolean = v == 0 && !sign_bit_set || v == 0 - 1 && sign_bit_set;
    if (done) {
      out = out.append(byte as u8);
      return out;
    }
    out = out.append((byte | 128) as u8);
  }
  return out;
}

pub function sleb_i64(buf: u8[], value: i64): u8[] {
  let out: u8[] = buf;
  let v: i64 = value;
  while (true) {
    let byte: i64 = v & 127i64;
    v = v >> 7;
    let sign_bit_set: boolean = (byte & 64i64) != 0i64;
    let done: boolean = v == 0i64 && !sign_bit_set || v == 0i64 - 1i64 && sign_bit_set;
    if (done) {
      out = out.append(byte as u8);
      return out;
    }
    out = out.append((byte | 128i64) as u8);
  }
  return out;
}

pub function uleb_size_u32(value: u32): i32 {
  let v: u32 = value;
  let n: i32 = 1;
  while (true) {
    v = v >> 7;
    if (v == 0u32) {
      return n;
    }
    n = n + 1;
  }
  return n;
}

pub function uleb_size_u64(value: u64): i32 {
  let v: u64 = value;
  let n: i32 = 1;
  while (true) {
    v = v >> 7;
    if (v == 0u64) {
      return n;
    }
    n = n + 1;
  }
  return n;
}
```
