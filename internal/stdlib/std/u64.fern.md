# Unsigned 64-bit integer methods

This module provides unsigned 64-bit integer methods and converts `usize` values to strings.

## Model

The module extends unsigned 64-bit values with arithmetic, bit operations, and decimal text. It also supplies decimal conversion for pointer-sized unsigned values.

## Contracts

Checked and saturation methods retain their distinct overflow policies. The width of `usize` follows the compilation target.

## Executable example

```fern test name=std-u64-example
import "std/u64";

function main(): i32 {
  let n: u64 = 42;
  if (n.to_string() != "42" || n.is_odd()) { return 1; }
  return 0;
}
```

pub function (n: u64) checked_pow(exp: i32): Option[u64] {
  if (exp < 0) {
    return None;
  }
  if (n == 0 as u64) {
    if (exp == 0) {
      return Some(1 as u64);
    }
    return Some(0 as u64);
  }
  let result: u64 = 1 as u64;
  let base: u64 = n;
  let e: i32 = exp;
  while (e > 0) {
    if (e % 2 == 1) {
      let p: u64 = result * base;
      if (p / base != result) {
        return None;
      }
      result = p;
    }
    e = e / 2;
    if (e > 0) {
      let q: u64 = base * base;
      if (q / base != base) {
        return None;
      }
      base = q;
    }
  }
  return Some(result);
}

    return n;

pub function (n: u64) log2_floor(): i32 {
  if (n == 0 as u64) {
    return 0 - 1;
  }
  let r: i32 = 0;
  let m: u64 = n;
  while (m > 1 as u64) {
    m = m / (2 as u64);
    r = r + 1;

}

pub function (n: u64) log10_floor(): i32 {

  while (m >= 10 as u64) {
    m = m / (10 as u64);
    r = r + 1;
  }
  return r;
}

pub function (n: u64) count_ones(): i32 {

pub function (n: u64) count_zeros(): i32 {

pub function (n: u64) rotate_left(bits: i32): u64 {
  let k: i32 = bits & 63;
  if (k == 0) {
    return n;
  }
  let left: u64 = n << k as u64;
  let right: u64 = n >> (64 - k) as u64;
  return left | right;
}

pub function (n: u64) rotate_right(bits: i32): u64 {
  let k: i32 = bits & 63;
  if (k == 0) {
    return n;
  }
  let right: u64 = n >> k as u64;
  let left: u64 = n << (64 - k) as u64;
  return right | left;
}

pub function (n: u64) byte_swap(): u64 {
  let r: u64 = 0 as u64;
  let i: i32 = 0;
  while (i < 8) {
    let b: u64 = n >> (i * 8) as u64 & 255 as u64;
    r = r | b << ((7 - i) * 8) as u64;
    i = i + 1;
  }
  return r;
}

pub function (n: u64) reverse_bits(): u64 {
  let u: u64 = n;
  let r: u64 = 0 as u64;
  let i: i32 = 0;
  while (i < 64) {
    r = r << 1 as u64 | u & 1 as u64;
    u = u >> 1 as u64;
    i = i + 1;
  }
  return r;
}

pub function (n: u64) bit(i: i32): boolean {
  if (i < 0 || i >= 64) {
    return false;
  }
  return (n & 1 as u64 << i as u64) != 0 as u64;
}

pub function (n: u64) set_bit(i: i32): u64 {
  if (i < 0 || i >= 64) {
    return n;

pub function (n: u64) clear_bit(i: i32): u64 {
  if (i < 0 || i >= 64) {
    return n;
  }
  let inv: u64 = (0 as i64 - 1 as i64) as u64 ^ 1 as u64 << i as u64;
  return n & inv;
}

pub function (n: u64) toggle_bit(i: i32): u64 {
  if (i < 0 || i >= 64) {

### `digits and number theory`

Digit and number-theory methods that `std/i32` already has, without the signed-only ones (`abs`, `signum`, `sign_str`, `is_minus_one`). `percent_of` never forms `n * 100`, so it is exact for every input.

  return (n / d, n % d);

pub function (n: u64) is_one(): boolean {
  return n == 1 as u64;

pub function (n: u64) is_perfect_square(): boolean {
  let k: u64 = n.sqrt_floor();
  return k * k == n;

pub function (n: u64) is_prime(): boolean {
  if (n < 2 as u64) {
    return false;
  }
  if (n < 4 as u64) {
    return true;
  }
  if (n % (2 as u64) == 0 as u64 || n % (3 as u64) == 0 as u64) {
    return false;
  }
  let i: u64 = 5 as u64;
  while (i <= n / i) {
    if (n % i == 0 as u64 || n % (i + 2 as u64) == 0 as u64) {
      return false;
    }
    i = i + 6 as u64;
  }
  return true;
}

pub function (n: u64) digits(): i32 {
  let m: u64 = n;
  let c: i32 = 1;
  while (m >= 10 as u64) {
    c = c + 1;
    m = m / (10 as u64);
  }
  return c;
}

pub function (n: u64) has_digit(d: i32): boolean {
  if (d < 0 || d > 9) {
    return false;
  }
  let m: u64 = n;
  while (true) {
    if ((m % (10 as u64)) as i32 == d) {
      return true;
    }
    m = m / (10 as u64);
    if (m == 0 as u64) {
      return false;
    }
  }
  return false;
}

pub function (n: u64) sum_of_digits(): i32 {
  let m: u64 = n;
  let s: i32 = 0;
  while (m > 0 as u64) {
    s = s + (m % (10 as u64)) as i32;
    m = m / (10 as u64);

  return s;

// Wraps when the reversed number does not fit.
pub function (n: u64) reverse_digits(): u64 {
  let m: u64 = n;
  let r: u64 = 0 as u64;
  while (m > 0 as u64) {
    r = r * (10 as u64) + m % (10 as u64);
    m = m / (10 as u64);

  return r;

pub function (n: u64) is_palindrome(): boolean {

// Zero above 20, where the result no longer fits.
pub function (n: u64) factorial(): u64 {
  if (n > 20 as u64) {
    return 0 as u64;
  }
  let result: u64 = 1 as u64;
  let i: u64 = 2 as u64;
  while (i <= n) {
    result = result * i;
    i = i + 1 as u64;
  }
  return result;
}

pub function (n: u64) ordinal(): string {
  let suffix: string = "th";
  let mod100: i32 = (n % (100 as u64)) as i32;
  if (mod100 < 11 || mod100 > 13) {
    let mod10: i32 = mod100 % 10;
    if (mod10 == 1) {
      suffix = "st";
    } else if (mod10 == 2) {
      suffix = "nd";
    } else if (mod10 == 3) {
      suffix = "rd";
    }
  }
  return n.to_string() + suffix;
}

pub function (n: u64) pluralize(singular: string, plural: string): string {
  if (n == 1 as u64) {
    return singular;
  }
  return plural;
}

// The whole percentage n is of total, truncated; 0 when total is 0.
// Exact for every input: n * 100 is never formed.
pub function (n: u64) percent_of(total: u64): u64 {
  if (total == 0 as u64) {
    return 0 as u64;
  }
  return n / total * (100 as u64) + __percent_below(n % total, total);
}

// floor(r * 100 / t) for r < t, without forming r * 100.
function __percent_below(r: u64, t: u64): u64 {
  if (t <= 184467440737095516 as u64) {
    return r * (100 as u64) / t;
  }
  // t = 100a + b, so k * t <= 100r exactly when k * b <= 100 (r - k * a).
  let a: u64 = t / (100 as u64);
  let b: u64 = t % (100 as u64);
  let k: u64 = r / a;
  if (k > 99 as u64) {
    k = 99 as u64;
  }
  while (k > 0 as u64) {
    if (k * a <= r) {
      let d: u64 = r - k * a;
      if (d >= 99 as u64 || k * b <= d * (100 as u64)) {
        return k;
      }
    }
    k = k - 1 as u64;
  }
  return 0 as u64;
}

// The largest multiple of m at or below n; n itself when m is 0.
pub function (n: u64) round_down_to(m: u64): u64 {
  if (m == 0 as u64) {
    return n;
  }
  return n - n % m;
}
