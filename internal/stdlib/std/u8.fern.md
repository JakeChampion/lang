# Bytes

This module provides the `u8` methods. The ASCII methods test and convert one byte without a Unicode interpretation. The module also parses a `u8` from text and writes one in a radix.

## Contracts

An ASCII test of a byte above 127 answers false. `to_ascii_lower` and `to_ascii_upper` change only the 26 ASCII letters.

`digit_value` and `hex_value` answer `None` for a byte that is not a digit of that radix.

## Executable example

```fern test name=std-u8-example
import "std/u8";

function main(): i32 {
  let b = 65 as u8;
  if (!b.is_ascii_letter() || b.to_ascii_lower() != 97 as u8) { return 1; }
  let Ok(n) = u8.parse_u8("255") else { return 2; };
  if (n.to_hex() != "ff") { return 3; }
  return 0;
}
```

### ASCII methods

```fern
<<ascii methods>>=
pub function (b: u8) is_ascii_digit(): boolean {
  return b >= 48 && b <= 57;
}

pub function (b: u8) is_ascii_alphanumeric(): boolean {
  return b.is_ascii_digit() || b.is_ascii_letter();
}

pub function (b: u8) is_ascii_whitespace(): boolean {
  return string.__is_ascii_ws(b as i32);
}

pub function (b: u8) is_ascii_newline(): boolean {
  return b == 10 || b == 13;
}

pub function (b: u8) is_ascii_vowel(): boolean {
  if (b == 97 || b == 101 || b == 105 || b == 111 || b == 117) {
    return true;
  }
  if (b == 65 || b == 69 || b == 73 || b == 79 || b == 85) {
    return true;
  }
  return false;
}

pub function (b: u8) is_ascii_printable(): boolean {
  return b >= 32 && b <= 126;
}

pub function (b: u8) is_ascii_control(): boolean {
  return b >= 0 && b <= 31 || b == 127;
}

pub function (b: u8) matches_any(bytes: string): boolean {
  let n = bytes.len();
  let i = 0;
  while (i < n) {
    if (bytes[i] == b) {
      return true;
    }
    i += 1;
  }
  return false;
}

pub function (b: u8) is_ascii_letter(): boolean {
  return b >= 65 && b <= 90 || b >= 97 && b <= 122;
}

pub function (b: u8) is_ascii_hex_digit(): boolean {
  return b.is_ascii_digit()
    || b >= 65 && b <= 70
    || b >= 97 && b <= 102;
}

pub function (b: u8) is_ascii(): boolean {
  return b >= 0 && b <= 127;
}

pub function (b: u8) is_ascii_punct(): boolean {
  if (b >= 33 && b <= 47) {
    return true;
  }
  if (b >= 58 && b <= 64) {
    return true;
  }
  if (b >= 91 && b <= 96) {
    return true;
  }
  if (b >= 123 && b <= 126) {
    return true;
  }
  return false;
}

pub function (b: u8) hex_digit(): string {
  if (b < 0 || b > 15) {
    return "";
  }
  let buf = __alloc_u8(1);
  if (b < 10) {
    buf = buf.with(0, (b + 48) as u8);
  } else {
    buf = buf.with(0, (b + 87) as u8);
  }
  unsafe {
    return string_from_bytes_unchecked(buf);
  }
}

pub function (b: u8) digit_value(): Option[i32] {
  if (b >= 48 && b <= 57) {
    return Some(b as i32 - 48);
  }
  return None;
}

pub function (b: u8) hex_value(): Option[i32] {
  if (b >= 48 && b <= 57) {
    return Some(b as i32 - 48);
  }
  if (b >= 97 && b <= 102) {
    return Some(b as i32 - 87);
  }
  if (b >= 65 && b <= 70) {
    return Some(b as i32 - 55);
  }
  return None;
}

pub function (b: u8) to_ascii_string(): string {
  if (b > 127) {
    return "";
  }
  let buf = __alloc_u8(1);
  buf = buf.with(0, b as u8);
  unsafe {
    return string_from_bytes_unchecked(buf);
  }
}

pub function (b: u8) is_ascii_lower(): boolean {
  return b >= 97 && b <= 122;
}

pub function (b: u8) is_ascii_upper(): boolean {
  return b >= 65 && b <= 90;
}

pub function (b: u8) to_ascii_lower(): u8 {
  if (b >= 65 && b <= 90) {
    return b + 32;
  }
  return b;
}

pub function (b: u8) to_ascii_upper(): u8 {
  if (b >= 97 && b <= 122) {
    return b - 32;
  }
  return b;
}
```

### Parsers and formats

```fern
<<parsers and formats>>=
pub function parse_u8(s: str): Result[u8, number_parse.ParseNumberError] {
  return parse_u8_radix(s, 10);
}

pub function parse_u8_radix(s: str, radix: i32): Result[u8, number_parse.ParseNumberError] {
  let (bits, code) = number_parse._integer(s, radix, false, 255 as u64, 0 as u64);
  if (code != number_parse.SUCCESS) {
    return Err(number_parse._error(code, radix));
  }
  return Ok(bits as u8);
}

pub function (n: u8) to_binary(): string {
  return n.to_string_radix(2);
}

pub function (n: u8) to_hex(): string {
  return n.to_string_radix(16);
}

pub function (n: u8) to_oct(): string {
  return n.to_string_radix(8);
}

pub function (n: u8) to_string_padded(width: i32): string {
  return int._pad_integer(n.to_string(), width);
}

pub function (n: u8) to_string_radix(base: i32): string {
  if (base == 10) {
    return n.to_string();
  }
  return int._format_integer(n as u64, false, base);
}

pub function (n: u8) to_string_with_sep(sep: string): string {
  return int._group_integer(n.to_string(), sep);
}
```
