# HTTP/2 connection state

This module implements an HTTP/2 connection as a state machine without I/O.

## Model

A connection step returns the next connection, output bytes, and protocol events. The caller supplies input bytes and performs the actual I/O.

## Contracts

Stream state and flow-control credit constrain response writes. A caller must retain the returned connection after each step and account for consumed input.

```fern
pub function (c: Conn) request(out: Outgoing): (Step, i32) {
  if (c.phase == CLOSED || c.peer_going_away || !c.client) {
    return (Step { conn: c, output: [], events: [] }, 0 - 1);
  }
  let stream: i32 = c.next_stream;
  let enc: hpack.Encoded = __request_block(c.encoder, out);
  let last: boolean = out.body.len() == 0 && out.trailers.len() == 0;
  let state: i32 = OPEN;
  if (last) {
    state = HALF_LOCAL;
  }
  let s: __Stream = __Stream { id: stream, state: state, send_window: c.peer_initial_window, recv_window: c.config.initial_window, method: out.method, scheme: out.scheme, authority: out.authority, path: out.path, fields: headers.header_map_new(), body: [], length: 0 - 1, handling: false, reset: false, pending: out.body, trailers: out.trailers, owes_end: !last, streaming: false, received: 0, producing: false, status: 0, seen: 0 };
  c = Conn { ...c, encoder: enc.encoder, streams: c.streams.append(s), last_stream: stream, next_stream: stream + 2 };
  let r: __Run = __header_frames(__run(c), stream, enc.bytes, last);
  if (!last) {
    r = __flushed(r);
  }
  return (__step(r), stream);
}
```

```fern
function __request_block(e: hpack.Encoder, out: Outgoing): hpack.Encoded {
  let en: hpack.Encoding = e.begin().put(":method", out.method, false).put(":scheme", out.scheme, false).put(":authority", out.authority, false).put(":path", out.path, false);
  let has_length: boolean = false;
  let i: i32 = 0;
  while (i < out.headers.names.len()) {
    let name: string = __lowered(out.headers.names[i]);
    if (name == "content-length") {
      has_length = true;
    }
    if (!__connection_specific(name) && name != "host") {
      en = en.put(name, out.headers.values[i], __sensitive(name));
    }
    i = i + 1;
  }
  if (!has_length && out.body.len() > 0) {
    en = en.put("content-length", int.int_to_string(out.body.len()), false);
  }
  return en.finish();
}
```

```fern
<<15 pub function (c: Conn) read(bytes: u8[]): Step>>=
pub function (c: Conn) read(bytes: u8[]): Step {
  if (c.phase == CLOSED) {
    return Step { conn: c, output: [], events: [] };
  }
  let held: boolean = c.inbox.len() > 0;
  let r: __Run = __run(Conn { ...c, inbox: __joined(c.inbox, bytes) });
  if (r.conn.phase == AWAITING_PREFACE) {
    r = __preface(r);
  }
  while (r.conn.phase == AWAITING_SETTINGS || r.conn.phase == RUNNING) {
    let before: i32 = r.conn.inbox.len() - r.conn.inbox_at;
    r = __next_frame(r);
    if (r.conn.inbox.len() - r.conn.inbox_at == before) {
      break;
    }
  }
  if (r.conn.inbox_at > 0 || !held && r.conn.inbox.len() > 0) {
    r = __Run { ...r, conn: Conn { ...r.conn, inbox: __rest(r.conn.inbox, r.conn.inbox_at), inbox_at: 0 } };
  }
  return __step(r);
}
```

### `16 function __preface(r: __Run): __Run`

```fern
<<16 function __preface(r: __Run): __Run>>=
function __preface(r: __Run): __Run {
  let want: u8[] = frame.preface();
  let have: u8[] = r.conn.inbox;
  let n: i32 = have.len();
  if (n > want.len()) {
    n = want.len();
  }
  let i: i32 = 0;
  while (i < n) {
    if (have[i] != want[i]) {
      return __fail(r, frame.PROTOCOL_ERROR, "not the HTTP/2 connection preface");
    }
    i = i + 1;
  }
  if (have.len() < want.len()) {
    return r;
  }
  return __Run { ...r, conn: Conn { ...r.conn, phase: AWAITING_SETTINGS, inbox_at: want.len() } };
}

function __next_frame(r: __Run): __Run {
  if let Some(plain) = frame.plain_headers_at(r.conn.inbox, r.conn.inbox_at, r.conn.config.max_frame_size) {
    if (r.conn.block_stream == 0 && r.conn.phase == RUNNING) {
      return __plain_headers(r, plain);
    }
  }
  match (frame.read(r.conn.inbox, r.conn.inbox_at, r.conn.config.max_frame_size)) {
    Ok(got) => {
      match (got) {
        frame.Got(f, n) => {
          let out: u8[] = r.out;
          let events: Event[] = r.events;
          let c: Conn = r.conn;
          c = __frame_taken(c, n);
          return __frame(__Run { conn: c, out: out, events: events }, f);
        },
        frame.More => {
          return r;
        }
      }
    },
    Err(e) => {
      if (e.stream == 0) {
        return __fail(r, e.code, e.reason);
      }
      let dropped: __Run = __Run { ...r, conn: Conn { ...r.conn, inbox_at: __past_frame(r.conn.inbox, r.conn.inbox_at) } };
      return __reset_stream(dropped, e.stream, e.code);
    }
  }
  return r;
}

function __plain_headers(r: __Run, len: i32): __Run {
  let at: i32 = r.conn.inbox_at;
  let stream: i32 = frame.stream_at(r.conn.inbox, at);
  let end_stream: boolean = frame.end_stream_at(r.conn.inbox, at);
  let from: i32 = at + frame.HEADER_LEN;
  return __block_done(__Run { ...r, conn: __frame_taken(r.conn, frame.HEADER_LEN + len) }, stream, true, from, from + len, end_stream);
}

function __frame_taken(c: Conn, n: i32): Conn {
  let frames: i32 = c.frames + 1;
  let resets: i32 = c.resets;
  let empty_data: i32 = c.empty_data;
  if (frames >= c.config.budget_frames) {
    frames = 0;
    resets = 0;
    empty_data = 0;
  }
  return Conn { ...c, inbox_at: c.inbox_at + n, frames: frames, resets: resets, empty_data: empty_data };
}

function __past_frame(inbox: u8[], at: i32): i32 {
  let len: i32 = inbox[at] as i32 << 16 | inbox[at + 1] as i32 << 8 | inbox[at + 2] as i32;
  return at + frame.HEADER_LEN + len;
}

function __frame(r: __Run, f: frame.Frame): __Run {
  if (r.conn.block_stream != 0) {
    return __in_block(r, f);
  }
  if (r.conn.phase == AWAITING_SETTINGS) {
    return __first_frame(r, f);
  }
  match (f) {
    frame.Data(d) => {
      return __data(r, d);
    },
    frame.Headers(h) => {
      return __headers(r, h);
    },
    frame.Settings(ack, settings) => {
      return __settings(r, ack, settings);
    },
    frame.WindowUpdate(stream, increment) => {
      return __window_update(r, stream, increment);
    },
    frame.RstStream(stream, code) => {
      return __rst_stream(r, stream, code);
    },
    _ => {
      return __frame_rest(r, f);
    }
  }
  return r;
}
```

### `17 function __frame_rest(r: __Run, f: frame.Frame): __Run`

```fern
<<17 function __frame_rest(r: __Run, f: frame.Frame): __Run>>=
function __frame_rest(r: __Run, f: frame.Frame): __Run {
  match (f) {
    frame.Ping(ack, data) => {
      if (ack) {
        return r;
      }
      return __send(r, frame.Ping(true, data));
    },
    frame.GoAway(g) => {
      return __Run { ...r, conn: Conn { ...r.conn, peer_going_away: true }, events: r.events.append(Event.GoAway(g.code)) };
    },
    frame.PushPromise(p) => {
      return __fail(r, frame.PROTOCOL_ERROR, "PUSH_PROMISE with push disabled");
    },
    frame.Continuation(stream, end, block) => {
      return __fail(r, frame.PROTOCOL_ERROR, "CONTINUATION with no header block open");
    },
    frame.PriorityFrame(stream, p) => {
      if (p.dependency == stream) {
        return __reset_stream(__Run { ...r, conn: Conn { ...r.conn, last_stream: __max(r.conn.last_stream, stream) } }, stream, frame.PROTOCOL_ERROR);
      }
      return r;
    },
    _ => {
      return r;
    }
  }
  return r;
}

function __self_dependent(priority: Option[frame.Priority], stream: i32): boolean {
  match (priority) {
    Some(p) => {
      return p.dependency == stream;
    },
    None => {
      return false;
    }
  }
  return false;
}

function __first_frame(r: __Run, f: frame.Frame): __Run {
  match (f) {
    frame.Settings(ack, settings) => {
      if (ack) {
        return __fail(r, frame.PROTOCOL_ERROR, "the first frame is a SETTINGS ACK");
      }
      return __settings(__Run { ...r, conn: Conn { ...r.conn, phase: RUNNING } }, false, settings);
    },
    _ => {
      return __fail(r, frame.PROTOCOL_ERROR, "the first frame is not SETTINGS");
    }
  }
  return r;
}

function __in_block(r: __Run, f: frame.Frame): __Run {
  match (f) {
    frame.Continuation(stream, end_headers, block) => {
      if (stream != r.conn.block_stream) {
        return __fail(r, frame.PROTOCOL_ERROR, "CONTINUATION on another stream");
      }
      return __block_part(r, block, end_headers);
    },
    _ => {
      return __fail(r, frame.PROTOCOL_ERROR, "a frame inside a header block");
    }
  }
  return r;
}

function __block_part(r: __Run, part: u8[], end_headers: boolean): __Run {
  let block: u8[] = __joined(r.conn.block, part);
  if (r.conn.block_frames + 1 > r.conn.config.header_frames_max || block.len() > r.conn.config.max_header_list + r.conn.config.max_frame_size) {
    return __fail(r, frame.ENHANCE_YOUR_CALM, "header block over the budget");
  }
  let out: u8[] = r.out;
  let events: Event[] = r.events;
  let c: Conn = r.conn;
  if (!end_headers) {
    c = Conn { ...c, block: block, block_frames: c.block_frames + 1 };
    return __Run { conn: c, out: out, events: events };
  }
  let stream: i32 = c.block_stream;
  let end_stream: boolean = c.block_end;
  let n: i32 = block.len();
  c = Conn { ...c, block_stream: 0, block: block, block_frames: 0 };
  let done: __Run = __block_done(__Run { conn: c, out: out, events: events }, stream, false, 0, n, end_stream);
  return __Run { ...done, conn: Conn { ...done.conn, block: [] } };
}

function __headers(r: __Run, h: frame.HeadersFrame): __Run {
  let out: u8[] = r.out;
  let events: Event[] = r.events;
  let c: Conn = r.conn;
  if (__self_dependent(h.priority, h.stream)) {
    c = Conn { ...c, last_stream: __max(c.last_stream, h.stream) };
    return __reset_stream(__Run { conn: c, out: out, events: events }, h.stream, frame.PROTOCOL_ERROR);
  }
  c = Conn { ...c, block_stream: h.stream, block: [], block_end: h.end_stream, block_frames: 0 };
  return __block_part(__Run { conn: c, out: out, events: events }, h.block, h.end_headers);
}

function __block_done(r: __Run, stream: i32, in_inbox: boolean, from: i32, end: i32, end_stream: boolean): __Run {
  let at: i32 = __stream_at(r.conn, stream);
  if (r.conn.client) {
    return __client_block(r, stream, at, in_inbox, from, end, end_stream);
  }
  if (at >= 0) {
    return __trailers(r, at, in_inbox, from, end, end_stream);
  }
  return __request(r, stream, in_inbox, from, end, end_stream);
}

function __skipped(r: __Run, in_inbox: boolean, from: i32, end: i32): __Run {
  let c: Conn = r.conn;
  let block: u8[] = __block_bytes(c, in_inbox);
  let st: hpack.Decoder = c.decoder.start_in(from, end);
  while (st.more()) {
    st = st.next(block);
  }
  return __decoded(__Run { ...r, conn: Conn { ...c, decoder: st } });
}

function __block_bytes(c: Conn, in_inbox: boolean): u8[] {
  if (in_inbox) {
    return c.inbox;
  }
  return c.block;
}
```

```fern
function __decoded(r: __Run): __Run {
  if (!r.conn.decoder.failed) {
    return r;
  }
  let why: string = __hpack_reason(r.conn.decoder.error);
  return __fail(r, frame.COMPRESSION_ERROR, why);
}

function __last_seen(last: i32, stream: i32, failed: boolean): i32 {
  if (failed) {
    return last;
  }
  return __max(last, stream);
}

function __hpack_reason(error: Option[hpack.HpackError]): string {
  match (error) {
    Some(e) => {
      return e.message();
    },
    None => {
      return "";
    }
  }
  return "";
}

function __reset_past(r: __Run, in_inbox: boolean, from: i32, end: i32, stream: i32, code: i32): __Run {
  let past: __Run = __skipped(r, in_inbox, from, end);
  if (past.conn.phase == CLOSED) {
    return past;
  }
  return __reset_stream(past, stream, code);
}

function __client_block(r: __Run, stream: i32, at: i32, in_inbox: boolean, from: i32, end: i32, end_stream: boolean): __Run {
  if (at < 0) {
    if (stream % 2 == 0) {
      return __fail(r, frame.PROTOCOL_ERROR, "HEADERS on a stream the server opened");
    }
    if (stream > r.conn.last_stream) {
      return __fail(r, frame.PROTOCOL_ERROR, "HEADERS on an idle stream");
    }
```

```fern
    n = 0;

    body = [];

    length = 0 - 1;

    if (n > 0) {
      let out: u8[] = r.out;

    return __step(__ended(r, at));

  let kept: __Stream = c.streams[at];
```

```fern
function __response_block(own out: u8[], e: hpack.Encoder, resp: HttpResponse, length: i32, date: string): hpack.Encoded {
  let en: hpack.Encoding = e.begin_in(out).put(":status", __status_text(resp.status), false);
  let i: i32 = 0;

  let has_date: boolean = false;
  while (i < resp.headers.names.len()) {
    let name: string = __lowered(resp.headers.names[i]);
    if (name == "content-length") {
      has_length = true;
    }
    if (name == "date") {
      has_date = true;
    }
    if (!__connection_specific(name)) {
      en = en.put(name, resp.headers.values[i], __sensitive(name));
    }
    i = i + 1;
  }
  if (!has_length && length >= 0 && !http.http_bodiless(resp.status)) {
    en = en.put("content-length", int.int_to_string(length), false);

    en = en.put("date", date, false);

function __status_text(status: i32): string {
  if (status == 200) {

function __sensitive(name: string): boolean {

function __lowered(name: string): string {

function __header_frames(r: __Run, stream: i32, block: u8[], end_stream: boolean): __Run {
  let max: i32 = r.conn.peer_max_frame;
  let first: i32 = block.len();
  if (first > max) {
    first = max;
  }
  let head: u8[] = block;
  if (first < block.len()) {
    head = __bytes_range(block, 0, first);
  }
  let out: __Run = __send_bytes(r, frame.headers_bytes(stream, end_stream, first == block.len(), head));

    if (to > block.len()) {
      to = block.len();
    }
    out = __send(out, frame.Continuation(stream, to == block.len(), __bytes_range(block, at, to)));
```

```fern
function __flush_stream(r: __Run, at: i32): __Run {
  let out: __Run = r;
  while (true) {
    let c: Conn = out.conn;
    let s: __Stream = c.streams[at];
    if (!s.owes_end) {
      return out;
    }
    if (s.pending.len() == 0) {
      if (s.producing) {
        return out;
      }
      if (s.trailers.len() == 0) {
        out = __send_bytes(out, frame.data_bytes(s.id, true, []));
      }
      return __ended(out, at);
    }
    let n: i32 = __min(__min(s.pending.len(), c.peer_max_frame), __min(s.send_window, c.send_window));
    if (n <= 0) {
      return out;
    }
    let final: boolean = n == s.pending.len() && s.trailers.len() == 0 && !s.producing;
    let sent: __Stream = __Stream { ...s, pending: __rest(s.pending, n), send_window: s.send_window - n };

    while (i < s.trailers.names.len()) {
      en = en.put(__lowered(s.trailers.names[i]), s.trailers.values[i], false);
      i = i + 1;

    let enc: hpack.Encoded = en.finish();
    let out: u8[] = r.out;
    let events: Event[] = r.events;
    let c: Conn = r.conn;
    c = Conn { ...c, encoder: enc.encoder };
    r = __header_frames(__Run { conn: c, out: out, events: events }, s.id, enc.bytes, true);
  }
  if (s.state == HALF_REMOTE) {
    return __Run { ...r, conn: Conn { ...r.conn, streams: __without(r.conn.streams, at) } };
  }
  let done: __Stream = __Stream { ...s, state: HALF_LOCAL, handling: false, pending: [], trailers: headers.header_map_new(), owes_end: false };
  return __Run { ...r, conn: Conn { ...r.conn, streams: r.conn.streams.with(at, done) } };
```
