# HTTP/3 session state

This module implements HTTP/3 protocol state without I/O.

## Model

A session handles HTTP/3 streams and produces protocol events without I/O. Control streams and request streams have different protocol roles.

## Contracts

The caller must route stream data to the session and deliver its output through a transport. HTTP/3 error codes retain their protocol-defined integer values.

  if (s.closed) {
    return s;

function (s: Session) __fail(code: i64, reason: string): Session {
  if (s.closed) {
    return s;
  }
  return Session { ...s, closed: true, actions: s.actions.append(CloseConnection(code, reason)) };
}

function (s: Session) __on_uni(id: i64, bytes: [u8], fin: boolean): Session {
  let i: i32 = s.__find_uni(id);
  let u: Uni = Uni { id: id, kind: 0 - 1 as i64, frames: __frames() };
  if (i >= 0) {
    u = s.unis[i];
  }
  let buf: u8[] = u.frames.buf.concat(__copy(bytes));
  let kept: Uni = Uni { ...u, frames: Frames { ...u.frames, buf: [] } };
  if (u.kind >= 0) {
    return s.__put_uni(i, kept).__uni_bytes(kept, buf, fin);
  }
  let read: i64 = frames.varint_at(buf, 0);
  if (read < 0) {
    return s.__put_uni(i, Uni { ...u, frames: Frames { ...u.frames, buf: buf } });
  }
  let typed: Uni = Uni { ...kept, kind: read };
  let cur: Session = s.__claim(typed).__put_uni(i, typed);
  if (cur.closed) {
    return cur;
  }
  return cur.__uni_bytes(typed, __slice_range(buf, frames.varint_end(buf, 0), buf.len()), fin);
}

function (s: Session) __claim(u: Uni): Session {
  if (u.kind == CONTROL_STREAM || u.kind == ENCODER_STREAM || u.kind == DECODER_STREAM) {
    if (s.__critical_of(u.kind) >= 0) {
      return s.__fail(H3_STREAM_CREATION_ERROR, "a second critical stream of one type");
    }
    return s.__with_critical(u.kind, u.id);
  }
  if (u.kind == PUSH_STREAM) {
    if (s.server) {
      return s.__fail(H3_STREAM_CREATION_ERROR, "a push stream from a client");
    }
    return s.__fail(H3_ID_ERROR, "a push stream without MAX_PUSH_ID");
  }
  return Session { ...s, actions: s.actions.append(Stop(u.id, H3_STREAM_CREATION_ERROR)) };
}

function (s: Session) __critical_of(kind: i64): i64 {
  if (kind == CONTROL_STREAM) {
    return s.peer_control;
  }
  if (kind == ENCODER_STREAM) {
    return s.peer_encoder;
  }
  return s.peer_decoder;
}

function (s: Session) __with_critical(kind: i64, id: i64): Session {
  if (kind == CONTROL_STREAM) {
    return Session { ...s, peer_control: id };
  }
  if (kind == ENCODER_STREAM) {
    return Session { ...s, peer_encoder: id };
  }
  return Session { ...s, peer_decoder: id };
}

function (s: Session) __uni_bytes(u: Uni, buf: u8[], fin: boolean): Session {
  if (u.kind != CONTROL_STREAM && u.kind != ENCODER_STREAM && u.kind != DECODER_STREAM) {
    return s;
  }
  if (fin) {
    return s.__fail(H3_CLOSED_CRITICAL_STREAM, "a critical stream ended");
  }
  if (u.kind == ENCODER_STREAM) {
    return s.__encoder_bytes(buf);
  }
  if (u.kind == DECODER_STREAM) {
    match (s.enc.on_decoder_stream(buf)) {
      Ok(e) => {
        return Session { ...s, enc: e };
      },
      Err(e) => {
        return s.__fail(e.code(), e.message());
      }
    }
  }
  return s.__control_bytes(u, buf);
}

function (s: Session) __encoder_bytes(buf: u8[]): Session {
  match (s.dec.on_encoder_stream(buf)) {
    Ok(d) => {
      let cur: Session = Session { ...s, dec: d }.__decoder_instructions();
      for r in cur.reqs {
        if (r.blocked && !r.gone) {
          cur = cur.__on_request(r.id, [], false);
        }
      }
      return cur;
    },
    Err(e) => {
      return s.__fail(e.code(), e.message());
    }
  }
}

function (s: Session) __decoder_instructions(): Session {
  let out: u8[] = s.dec.out;
  if (out.len() == 0 || s.decoder_stream < 0) {
    return s;
  }
  return Session { ...s, dec: s.dec.instructions_taken(), actions: s.actions.append(Write(s.decoder_stream, out, false)) };
}

function (s: Session) __control_bytes(u: Uni, buf: u8[]): Session {
  let cur: Session = s;
  let f: Frames = Frames { ...u.frames, buf: buf };
  while (!cur.closed) {
    match (__next_frame(f)) {
      Ok(r) => {
        f = r.frames;
        if (r.kind < 0) {
          return cur.__put_uni(cur.__find_uni(u.id), Uni { ...u, frames: f });
        }
        cur = cur.__control_frame(r.kind, r.payload);
      },
      Err(why) => {
        return cur.__fail(H3_FRAME_ERROR, why);
      }
    }
  }
  return cur;
}

function (s: Session) __control_frame(kind: i64, payload: u8[]): Session {
  if (!s.settings_seen && kind != SETTINGS) {
    return s.__fail(H3_MISSING_SETTINGS, "the control stream did not open with SETTINGS");
  }
  if (kind == SETTINGS) {
    if (s.settings_seen) {
      return s.__fail(H3_FRAME_UNEXPECTED, "a second SETTINGS");
    }
    return s.__settings(payload);
  }
  if (kind == GOAWAY) {
    return s.__on_goaway(payload);
  }
  if (__control_forbids(kind, s.server)) {
    return s.__fail(H3_FRAME_UNEXPECTED, "a frame not allowed on the control stream");
  }
  return s;
}

function __control_forbids(kind: i64, server: boolean): boolean {
  return kind == MAX_PUSH_ID && !server || kind == DATA || kind == HEADERS || kind == PUSH_PROMISE || __http2_frame(kind);
}

function __http2_frame(kind: i64): boolean {
  return kind == 2 || kind == 6 || kind == 8 || kind == 9;
}

function (s: Session) __settings(payload: u8[]): Session {
  let p: Settings = s.peer;
  let seen: i64[] = [];
  let at: i32 = 0;
  while (at < payload.len()) {
    let k: i64 = frames.varint_at(payload, at);
    if (k < 0) {
      return s.__fail(H3_FRAME_ERROR, "a truncated setting");
    }
    let vat: i32 = frames.varint_end(payload, at);
    let v: i64 = frames.varint_at(payload, vat);
    if (v < 0) {
      return s.__fail(H3_FRAME_ERROR, "a truncated setting");
    }
    if (seen.contains(k) || k >= 2 && k <= 5) {
      return s.__fail(H3_SETTINGS_ERROR, "a repeated setting, or one of HTTP/2's");
    }
    seen = seen.append(k);
    p = __setting(p, k, v);
    at = frames.varint_end(payload, vat);
  }
  return Session { ...s, peer: p, settings_seen: true };
}

function __setting(p: Settings, id: i64, v: i64): Settings {
  if (id == SETTING_QPACK_MAX_TABLE_CAPACITY) {
    return Settings { ...p, max_table_capacity: v };
  }
  if (id == SETTING_QPACK_BLOCKED_STREAMS) {
    return Settings { ...p, blocked_streams: v };
  }
  if (id == SETTING_MAX_FIELD_SECTION_SIZE) {
    return Settings { ...p, max_field_section_size: v };
  }
  return p;
}

function (s: Session) __on_goaway(payload: u8[]): Session {
  let id: i64 = frames.varint_at(payload, 0);
  if (id < 0) {
    return s.__fail(H3_FRAME_ERROR, "a truncated GOAWAY");
  }
  if (s.server) {
    return s;
  }
  if (id % 4 != 0 || s.goaway >= 0 && id > s.goaway) {
    return s.__fail(H3_ID_ERROR, "a GOAWAY stream ID that is not a request stream's, or that rose");
  }
  let cur: Session = Session { ...s, goaway: id, events: s.events.append(GoawayReceived(id)) };
  for q in s.reqs {
    if (q.id >= id && !q.gone) {
      cur = Session { ...cur, events: cur.events.append(StreamAborted(q.id, H3_REQUEST_REJECTED)) }.__forget(q.id);
    }
  }
  return cur;
}

function __req(id: i64): Req {
  return Req { id: id, frames: __frames(), phase: BEFORE_HEADERS, held: [], blocked: false, length: 0 - 1 as i64, received: 0 as i64, fin: false, done: false, sent_fin: false, head: false, gone: false };
}

function (s: Session) __on_request(id: i64, bytes: [u8], fin: boolean): Session {
  let i: i32 = s.__find(id);
  let cur: Session = s;
  if (i < 0) {
    if (!s.server || (id & 1) != 0) {
      return s;
    }
    cur = s.__added(__req(id));
    i = cur.reqs.len() - 1;
  }
  let r: Req = cur.reqs[i];
  r = Req { ...r, frames: Frames { ...r.frames, buf: r.frames.buf.concat(__copy(bytes)) }, fin: r.fin || fin };
  return cur.__put_req(i, r).__request_frames(id);
}

function (s: Session) __request_frames(id: i64): Session {
  let cur: Session = s;
  while (!cur.closed) {
    let i: i32 = cur.__find(id);
    if (i < 0) {
      return cur;
    }
    let r: Req = cur.reqs[i];
    if (r.blocked) {
      return cur.__retry_held(i);
    }
    if (r.frames.data_left > 0 && r.frames.buf.len() > 0) {
      cur = cur.__body_bytes(i);
      continue;
    }
    match (__next_frame(r.frames)) {
      Ok(f) => {
        cur = cur.__put_req(i, Req { ...r, frames: f.frames });
        if (f.kind < 0) {
          return cur.__request_end(id);
        }
        cur = cur.__request_frame(id, f.kind, f.payload);
      },
      Err(why) => {
        return cur.__fail(H3_FRAME_ERROR, why);
      }
    }
  }
  return cur;
}

function (s: Session) __body_bytes(i: i32): Session {
  let r: Req = s.reqs[i];
  let n: i32 = r.frames.buf.len();
  if (r.frames.data_left < n as i64) {
    n = r.frames.data_left as i32;
  }
  let data: u8[] = __slice_range(r.frames.buf, 0, n);
  let f: Frames = Frames { ...r.frames, buf: __slice_range(r.frames.buf, n, r.frames.buf.len()), data_left: r.frames.data_left - n as i64 };
  let next: Req = Req { ...r, frames: f, received: r.received + n as i64 };
  if (next.length >= 0 && next.received > next.length) {
    return s.__malformed(r.id, "more body than its content-length");
  }
  return Session { ...s, events: s.events.append(DataReceived(r.id, data)) }.__put_req(i, next);
}

function (s: Session) __request_frame(id: i64, kind: i64, payload: u8[]): Session {
  let i: i32 = s.__find(id);
  let r: Req = s.reqs[i];
  if (kind == HEADERS) {
    if (r.phase == AFTER_TRAILERS) {
      return s.__fail(H3_FRAME_UNEXPECTED, "a frame after the trailers");
    }
    return s.__put_req(i, Req { ...r, held: payload, blocked: true }).__retry_held(i);
  }
  if (kind == DATA) {
    if (r.phase != IN_BODY) {
      return s.__fail(H3_FRAME_UNEXPECTED, "DATA outside a message's body");
    }
    return s;
  }
  return s.__request_other(kind);
}

function (s: Session) __request_other(kind: i64): Session {
  if (kind == PUSH_PROMISE && !s.server) {
    return s.__fail(H3_ID_ERROR, "PUSH_PROMISE without MAX_PUSH_ID");
  }
  if (kind == SETTINGS || kind == GOAWAY || kind == MAX_PUSH_ID || kind == CANCEL_PUSH || kind == PUSH_PROMISE || __http2_frame(kind)) {
    return s.__fail(H3_FRAME_UNEXPECTED, "a frame not allowed on a request stream");
  }
  return s;
}

function (s: Session) __retry_held(i: i32): Session {
  let r: Req = s.reqs[i];
  let cur: Session = Session { ...s, dec: s.dec.decode(r.id, r.held) };
  if let Some(e) = cur.dec.failed {
    return cur.__fail(e.code(), e.message());
  }
  if (cur.dec.waiting) {
    return cur;
  }
  let fields: qpack.Field[] = cur.dec.fields;
  cur = cur.__put_req(i, Req { ...r, held: [], blocked: false }).__decoder_instructions();
  return cur.__section(r.id, fields).__request_frames(r.id);
}

function (s: Session) __section(id: i64, fields: qpack.Field[]): Session {
  let i: i32 = s.__find(id);
  if (i < 0) {
    return s;
  }
  let r: Req = s.reqs[i];
  if (r.phase == IN_BODY) {
    if (__has_pseudo(fields)) {
      return s.__malformed(id, "a pseudo-header field in trailers");
    }

    return s;
